021|def dark_correct(light_img: np.ndarray, dark_img: np.ndarray) -> np.ndarray:
022|    """
023|    暗场校正：亮场减暗场，消除传感器底噪与固定杂反射
024|    :param light_img: 亮场原始图
025|    :param dark_img: 暗场原始图
026|    :return: 校正后有效灰度图
027|    """
028|    corrected = light_img - dark_img
029|    corrected[corrected < 0] = 0  # 剔除负向噪声
030|    return corrected
031|def select_roi(raw_image: np.ndarray) -> np.ndarray:
032|    """
033|    手动框选有效发光区域（回车确认，ESC取消）
034|    :param raw_image: 校正后灰度图
035|    :return: 裁切后的ROI区域
036|    """
037|    # 归一化显示，避免12bit图预览全黑
038|    display_img = cv2.normalize(raw_image, None, 0, 65535, cv2.NORM_MINMAX, dtype=cv2.CV_16U)
039|    roi = cv2.selectROI("框选有效发光区域（回车确认，ESC取消）", display_img, showCrosshair=True)
040|    cv2.destroyAllWindows()
041|    x, y, w, h = roi
042|    if w == 0 or h == 0:
043|        raise ValueError("未框选有效区域")
044|    return raw_image[y:y+h, x:x+w]
045|def calc_block_ki(
046|        roi_img: np.ndarray,
047|        row_num: int = 5,
048|        col_num: int = 12,
049|        ki_min: float = 0.6,
050|        ki_max: float = 1.5
051|) -> tuple[list[float], list[float]]:
052|    """
053|    划分网格分区，计算每个分区的平均灰度与补偿系数Ki
054|    :param roi_img: 裁切后的有效区域图像
055|    :param row_num: 分区行数
056|    :param col_num: 分区列数
057|    :param ki_min: Ki系数下限
058|    :param ki_max: Ki系数上限
059|    :return: (Ki系数列表, 分区平均灰度列表)，行优先顺序
060|    """
061|    h, w = roi_img.shape
062|    block_w = w // col_num
063|    block_h = h // row_num
064|    block_avg_list = []
065|    # 逐行逐列遍历分区
066|    for r in range(row_num):
067|        for c in range(col_num):
068|            x1 = c * block_w
069|            y1 = r * block_h
070|            x2 = x1 + block_w
071|            y2 = y1 + block_h
072|            block = roi_img[y1:y2, x1:x2]
073|            block_avg_list.append(float(np.mean(block)))
074|    # 全场基准平均灰度
075|    global_avg = float(np.mean(block_avg_list))
076|    ki_list = []
077|    for avg_val in block_avg_list:
078|        if avg_val < 1e-6:
079|            ki = 1.0
080|        else:
081|            ki = global_avg / avg_val
082|        # 硬限幅，防止硬件异常导致极端补偿
083|        ki = float(np.clip(ki, ki_min, ki_max))
084|        ki_list.append(round(ki, 4))
085|    return ki_list, block_avg_list

196|准备图片
197|暗场图（关灯拍摄）命名为 dark_12bit.tif
198|亮场图（开灯全屏纯白拍摄）命名为 light_12bit.tif

203|关键注意事项

205|标定全程相机曝光、增益、光圈、偏振角度必须固定，不得中途改动
206|图片必须是原生 12bit RAW 灰度图，禁止使用 JPG/PNG 等压缩格式
207|若更换任何光学元件（镜头、滤光片、偏振片、匀光板），必须重新拍摄暗场 + 亮场图再计算
208|合格标准：3D 曲面平滑无周期性波浪、Ki 系数大多集中在 0.85~1.15 区间、无极端限值
